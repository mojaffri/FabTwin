cmake_minimum_required(VERSION 3.16)
project(fabtwin_controller LANGUAGES CXX)
add_library(fabtwin_controller SHARED src/bridge.cpp)
target_include_directories(fabtwin_controller PRIVATE include)
target_compile_features(fabtwin_controller PRIVATE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
    add_executable(controller_tests tests/core_tests.cpp)
    target_include_directories(controller_tests PRIVATE include)
    target_compile_features(controller_tests PRIVATE cxx_std_17)
    add_test(NAME controller_core COMMAND controller_tests)
endif()
